This identity is why cross-entropy through a softmax collapses to p - y.

def attention(q, k, v, mask=None):
    scores = q @ k.transpose(-2, -1) / math.sqrt(q.shape[-1])
    if mask is not None:
        scores = scores + mask
    return torch.softmax(scores, dim=-1) @ v
